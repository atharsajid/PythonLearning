# To Install virtualenv package
pip3 install virtualenv

# To Create Virtual Env .venv is name of the environment
python3 -m venv .venv

# To Activate in Mac
source .venv/bin/activate

# To Check Python Version
python --version

# To Install new packages int the Environment
pip install requests
pip install pymongo
pip install textual

# To Check Installed Packages
pip list

# To Create requirements.txt
pip list > requirements.text


# To Deactivate
deactivate
